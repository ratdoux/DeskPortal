cmake_minimum_required(VERSION 3.21)
project(DeskCapture LANGUAGES CXX)
set(FFMPEG_PACKAGE "ffmpeg-n8.1.3-39-g660b5aa7db-win64-lgpl-shared-8.1")
set(FFMPEG_SHA256 8e424751cc51a473472c5bd99a79dade55792688917762295c188a844c5eca2a)
set(SDK_CACHE "${CMAKE_CURRENT_BINARY_DIR}/codec-sdk")
set(FFMPEG_ROOT "${SDK_CACHE}/${FFMPEG_PACKAGE}" CACHE PATH "FFmpeg 8.1 shared x64 SDK")
# Keep the pinned archive with the source so a clean checkout outlives upstream's daily builds.
set(FFMPEG_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/${FFMPEG_PACKAGE}.zip" CACHE FILEPATH "Pinned FFmpeg SDK archive")
if(NOT EXISTS "${FFMPEG_ROOT}/include/libavcodec/avcodec.h")
  if(EXISTS "${FFMPEG_ARCHIVE}")
    file(SHA256 "${FFMPEG_ARCHIVE}" archive_hash)
    if(NOT archive_hash STREQUAL FFMPEG_SHA256)
      message(FATAL_ERROR "${FFMPEG_ARCHIVE} does not match the pinned FFmpeg SDK checksum.")
    endif()
  else()
    get_filename_component(archive_folder "${FFMPEG_ARCHIVE}" DIRECTORY)
    file(MAKE_DIRECTORY "${archive_folder}")
    file(DOWNLOAD "https://github.com/FFmpeg-Builds/FFmpeg-Builds/releases/download/autobuild-2026-11-23-12-43/${FFMPEG_PACKAGE}.zip"
      "${FFMPEG_ARCHIVE}" EXPECTED_HASH SHA256=${FFMPEG_SHA256} TLS_VERIFY ON)
  endif()
  file(MAKE_DIRECTORY "${SDK_CACHE}")
  file(ARCHIVE_EXTRACT INPUT "${FFMPEG_ARCHIVE}" DESTINATION "${SDK_CACHE}")
endif()
add_library(DeskCapture SHARED Capture.cpp Video.cpp WindowIcon.cpp)
target_compile_features(DeskCapture PRIVATE cxx_std_20)
target_compile_definitions(DeskCapture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_property(TARGET DeskCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(DeskCapture PRIVATE "${FFMPEG_ROOT}/include")
target_link_libraries(DeskCapture PRIVATE d3d11 dxgi windowsapp ole32 shell32 propsys
  "${FFMPEG_ROOT}/lib/avcodec.lib" "${FFMPEG_ROOT}/lib/avutil.lib" "${FFMPEG_ROOT}/lib/swscale.lib")
foreach(dll avcodec-62 avutil-60 swscale-9 swresample-6)
  configure_file("${FFMPEG_ROOT}/bin/${dll}.dll" "codec-runtime/${dll}.dll" COPYONLY)
endforeach()
configure_file("${FFMPEG_ROOT}/LICENSE.txt" "codec-runtime/FFmpeg-LICENSE.txt" COPYONLY)
